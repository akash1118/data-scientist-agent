# requirements.txt
# -----------------
# WHY: This file lists every Python package the app needs, so anyone can
#      recreate the exact same environment with "pip install -r requirements.txt".
# WHAT: Streamlit (UI), LangGraph + LangChain (agent orchestration),
#       Google Gemini SDK (the LLM), Pandas/Matplotlib/Plotly (data + charts),
#       FAISS (vector search for RAG), python-dotenv (reads the .env file).

# --- Web UI ---
streamlit>=1.40.2

# --- Agentic AI / LLM orchestration ---
# These are pinned exactly (rather than "minimum version") because the code
# in agents/ and graph/ was written and tested against these specific APIs.
langgraph==1.2.9
langchain==1.3.14
langchain-core==1.5.1
langchain-community==0.4.2
langchain-google-genai==4.3.1

# Groq is wired in as an automatic FALLBACK chat provider (used only if
# Gemini fails and GROQ_API_KEY is set in .env) - see utils/llm.py get_llm().
langchain-groq==1.1.3

# --- RAG embeddings (local, open-source, no API key needed) ---
# HuggingFaceEmbeddings runs a small "sentence-transformers" model directly
# on your machine (CPU) instead of calling a paid embedding API. This also
# pulls in `sentence-transformers` and `torch` (CPU build), so the first
# `pip install` here will take longer and download a few hundred MB.
langchain-huggingface>=0.1.2
sentence-transformers>=3.0.0
# torchvision isn't actually used (we only embed text, never images), but a
# newer `transformers` version eagerly imports an image-processing submodule
# that requires torchvision just to be importable - without it, loading the
# embedding model crashes with "ModuleNotFoundError: No module named 'torchvision'".
torchvision>=0.19.0

# --- Data handling ---
# Minimum versions only (no upper pin) - this lets pip pick whichever
# version already has a pre-built "wheel" for YOUR Python version, instead
# of trying to compile an old version from source (which can fail on very
# new Python releases). Tested with Python 3.12 and 3.14.
pandas>=2.2.3
numpy>=2.1.3

# --- Charts ---
matplotlib>=3.10.0
plotly>=5.24.1

# --- RAG / vector search ---
faiss-cpu>=1.9.0.post1

# --- Report export ---
fpdf2>=2.8.2

# --- LLMOps demo: API Development + Model Serving (see llmops/api_server.py) ---
# A small FastAPI app + its server, plus `requests` to call it live from the
# Streamlit "LLMOps & Production" page.
fastapi>=0.115.0
uvicorn>=0.30.0
requests>=2.31.0

# --- Environment variables ---
python-dotenv>=1.0.1
